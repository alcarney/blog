   def evaluate(form, env):
       match form:
           case S(name):
               return env[name]
           case []:
               return []
           case [*fs]:
               f, *args = [evaluate(f, env) for f in fs]
               return f(*args)
           case _:
               return form

**rep**

.. code-block:: python


   env = {
     '+': lambda a,b: a+b,
     '-': lambda a,b: a-b,
     '*': lambda a,b: a*b,
     '/': lambda a,b: int(a/b)
   }


   def rep(ins: str):
       try:
           return printit(evaluate(read(ins), env))
       except EOFError:
           return f"EOF"
       except KeyError as e:
           return f"undefined symbol {e.args[0]!r}"
